"""THROWAWAY spike: can Playwright read Frontier PHL->MCO fares? Not app code."""
import asyncio, json, random, sys
sys.path.insert(0, "/Users/ahmetyesilbas/avelo-finder")
import config
from playwright.async_api import async_playwright

OUT = "/private/tmp/claude-501/-Users-ahmetyesilbas-Desktop-avelo-finder/e3a92706-91b7-47db-8628-0eaaabe88833/scratchpad/frontier"
URL = ("https://booking.flyfrontier.com/Flight/InternalSelect?o1=PHL&d1=MCO"
       "&dd1=Nov%2012%2C%202026&ADT=1&mon=true&promo=")


async def probe(headless: bool) -> dict:
    tag = "headless" if headless else "headful"
    async with async_playwright() as pw:
        b = await pw.chromium.launch(headless=headless, args=list(config.BROWSER_ARGS))
        ctx = await b.new_context(user_agent=random.choice(config.USER_AGENTS),
                                  viewport=dict(zip(("width", "height"), (1440, 900))), locale="en-US")
        await ctx.add_init_script(config.STEALTH_INIT_SCRIPT)
        page = await ctx.new_page()
        xhr = []
        page.on("response", lambda r: xhr.append((r.status, r.url[:140], r.headers.get("content-type", "")))
                if r.request.resource_type in ("xhr", "fetch", "document") else None)
        res = {"mode": tag}
        try:
            # land on home page first like a human, then the search URL
            r0 = await page.goto("https://www.flyfrontier.com/", wait_until="domcontentloaded", timeout=45000)
            res["home_status"] = r0.status if r0 else None
            await page.wait_for_timeout(4000)
            r = await page.goto(URL, wait_until="domcontentloaded", timeout=60000)
            res["search_status"] = r.status if r else None
            await page.wait_for_timeout(12000)
            html = await page.content()
            res["final_url"] = page.url
            res["title"] = await page.title()
            low = html.lower()
            res["blocked_markers"] = [m for m in ("px-captcha", "press & hold", "perimeterx", "access denied",
                                                  "_incapsula", "captcha", "are you a robot", "akamai")
                                      if m in low]
            res["fare_markers"] = [m for m in ("discount den", "den", "standard", "$") if m in low][:4]
            res["dollar_count"] = html.count("$")
            res["html_len"] = len(html)
            open(f"{OUT}_{tag}.html", "w").write(html)
            await page.screenshot(path=f"{OUT}_{tag}.png", full_page=False)
        except Exception as e:  # spike: record and move on
            res["error"] = repr(e)[:300]
        res["xhr"] = [x for x in xhr if "flyfrontier" in x[1]][:25]
        await b.close()
        return res


async def main():
    modes = [True] if "--headful" not in sys.argv else [False]
    for h in modes:
        print(json.dumps(await probe(h), indent=1))

asyncio.run(main())
